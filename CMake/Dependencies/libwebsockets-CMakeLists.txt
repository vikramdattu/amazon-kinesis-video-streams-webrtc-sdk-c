cmake_minimum_required(VERSION 3.6.3)

project(libwebsocket-download NONE)

include(ExternalProject)
if (BUILD_STATIC_LIBS)
  set(LWS_WITH_SHARED 0)
else()
  set(LWS_WITH_SHARED 1)
  if (USE_OPENSSL AND NOT WIN32)
    find_package(OpenSSL REQUIRED)
  endif()
endif()

if (USE_OPENSSL)
  set(LWS_WITH_MBEDTLS OFF)
else()
  set(LWS_WITH_MBEDTLS ON)
endif()

if (WIN32)
  set(LWS_WITH_THREADPOOL ${LWS_WITH_THREADPOOL})
  set(LWS_EXT_PTHREAD_INCLUDE_DIR ${LWS_EXT_PTHREAD_INCLUDE_DIR})
  set(LWS_EXT_PTHREAD_LIBRARIES ${LWS_EXT_PTHREAD_LIBRARIES})
  set(LWS_OPENSSL_INCLUDE_DIRS ${LWS_OPENSSL_INCLUDE_DIRS})
  set(LWS_OPENSSL_LIBRARIES ${LWS_OPENSSL_LIBRARIES})
endif()

ExternalProject_Add(project_libwebsockets
    GIT_REPOSITORY    https://github.com/warmcat/libwebsockets.git
    # TODO: revert this set of workarounds once warmcat cuts a release that
    # includes the mbedtls4 migration. Current state:
    #   - SHA pin to main 7726361 (post-mbedtls4-migration: 3079d311 + sai-fixes)
    #   - GIT_SHALLOW FALSE because the SHA is not a branch tip / tag
    #   - GIT_SUBMODULES "README.md" sidesteps broken qpack/qifs gitlink that
    #     has no .gitmodules URL entry (tracked: warmcat upstream PR pending,
    #     see esp-rainmaker task #130)
    #   - LWS_ROLE_QUIC=OFF / LWS_WITH_HTTP3=OFF skip in-flight features that
    #     don't compile cleanly (tracked: warmcat upstream PR pending,
    #     see esp-rainmaker task #131)
    # v4.5.8 (latest release tag) predates the mbedtls4 migration. Revert to
    # GIT_TAG vX.Y.Z + GIT_SHALLOW TRUE + drop GIT_SUBMODULES + drop the
    # LWS_ROLE_QUIC/LWS_WITH_HTTP3 lines once the upstream release ships.
    GIT_TAG           7726361cf453ec7bb83eb33ec6d5f468219c1c4d
    GIT_PROGRESS      TRUE
    # GIT_SHALLOW FALSE because the SHA above is not currently a branch tip on
    # warmcat/libwebsockets:main. Depth-1 clone would fail with "reference is
    # not a tree". Trade-off: ~80 MB full clone vs reproducibility of the
    # pinned SHA. Revert to TRUE once a release tag carrying the mbedtls4
    # migration ships.
    GIT_SHALLOW       FALSE
    # Skip libwebsockets's nested submodules — KVS doesn't use the example /
    # test data, and recent main SHAs commit the api-test-qpack/qifs gitlink
    # without a corresponding .gitmodules URL entry which breaks recursive
    # submodule init. Point GIT_SUBMODULES at an existing non-submodule path
    # (README.md) so `git submodule update --init -- <path>` is a no-op on
    # CMake versions where the empty-list form doesn't take.
    GIT_SUBMODULES    "README.md"
    PREFIX            ${CMAKE_CURRENT_BINARY_DIR}/build
    LIST_SEPARATOR    |
    CMAKE_ARGS
        -DCMAKE_INSTALL_PREFIX=${OPEN_SRC_INSTALL_PREFIX}
        -DCMAKE_C_FLAGS=${CMAKE_C_FLAGS}
        -DCMAKE_BUILD_TYPE=${CMAKE_BUILD_TYPE}
        -DDISABLE_WERROR=ON
        -DLWS_WITH_HTTP2=1
        -DLWS_ROLE_QUIC=OFF
        -DLWS_WITH_HTTP3=OFF
        -DLWS_HAVE_HMAC_CTX_new=1
        -DLWS_HAVE_SSL_EXTRA_CHAIN_CERTS=1
        -DLWS_HAVE_OPENSSL_ECDH_H=1
        -DLWS_HAVE_EVP_MD_CTX_free=1
        -DLWS_WITHOUT_SERVER=1
        -DLWS_WITHOUT_TESTAPPS=1
        -DLWS_WITHOUT_TEST_SERVER_EXTPOLL=1
        -DLWS_WITHOUT_TEST_PING=1
        -DLWS_WITHOUT_TEST_CLIENT=1
        -DLWS_STATIC_PIC=1
        -DLWS_WITH_STATIC=1
        -DLWS_WITH_SHARED=${LWS_WITH_SHARED}
        -DLWS_WITH_MBEDTLS=${LWS_WITH_MBEDTLS}
        -DLWS_MBEDTLS_LIBRARIES=${LWS_MBEDTLS_LIBRARIES}
        -DLWS_MBEDTLS_INCLUDE_DIRS=${LWS_MBEDTLS_INCLUDE_DIRS}
        -DLWS_WITH_MINIMAL_EXAMPLES=0
        -DLWS_HAVE_PTHREAD_H=1
        -DLWS_WITH_THREADPOOL=${LWS_WITH_THREADPOOL}
        -DLWS_WITH_ZLIB=0
        -DLWS_HAVE_EVENTFD=0
        -DLWS_EXT_PTHREAD_INCLUDE_DIR=${LWS_EXT_PTHREAD_INCLUDE_DIR}
        -DLWS_EXT_PTHREAD_LIBRARIES=${LWS_EXT_PTHREAD_LIBRARIES}
        -DLWS_OPENSSL_INCLUDE_DIRS=${LWS_OPENSSL_INCLUDE_DIRS}
        -DLWS_OPENSSL_LIBRARIES=${LWS_OPENSSL_LIBRARIES}
        -DLWS_WITH_IPV6=ON
    BUILD_ALWAYS      TRUE
    TEST_COMMAND      ""
)
